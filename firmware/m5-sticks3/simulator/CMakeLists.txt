cmake_minimum_required(VERSION 3.16)
project(m5stack_voice_ui_sim LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LVGL_SOURCE_DIR "" CACHE PATH "Existing LVGL 8.4.0 checkout (optional)")
if(NOT LVGL_SOURCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../components/lvgl/src/lv_conf_internal.h")
    get_filename_component(LVGL_SOURCE_DIR "../components/lvgl" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(NOT LVGL_SOURCE_DIR)
    include(FetchContent)
    FetchContent_Declare(lvgl_source GIT_REPOSITORY https://github.com/lvgl/lvgl.git GIT_TAG v8.4.0 GIT_SHALLOW TRUE)
    FetchContent_GetProperties(lvgl_source)
    if(NOT lvgl_source_POPULATED)
        FetchContent_Populate(lvgl_source)
    endif()
    set(LVGL_SOURCE_DIR "${lvgl_source_SOURCE_DIR}")
endif()
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_SOURCE_DIR}/src/*.c")
add_library(voice_lvgl STATIC ${LVGL_SOURCES})
target_include_directories(voice_lvgl PUBLIC "${LVGL_SOURCE_DIR}" ../main)
target_compile_definitions(voice_lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE LV_KCONFIG_IGNORE)
add_executable(voice_ui_sim simulator.cpp ../main/voice_ui.cpp
    ../main/voice_ui_font_12.c ../main/voice_ui_font_18.c)
target_compile_definitions(voice_ui_sim PRIVATE VOICE_UI_SIMULATOR)
target_link_libraries(voice_ui_sim PRIVATE voice_lvgl m)
target_compile_options(voice_ui_sim PRIVATE -Wall -Wextra)

add_executable(voice_ui_multiboard_smoke multiboard_smoke.cpp ../main/voice_ui.cpp
    ../main/voice_ui_font_12.c ../main/voice_ui_font_18.c)
target_compile_definitions(voice_ui_multiboard_smoke PRIVATE VOICE_UI_SIMULATOR)
target_link_libraries(voice_ui_multiboard_smoke PRIVATE voice_lvgl m)
target_compile_options(voice_ui_multiboard_smoke PRIVATE -Wall -Wextra)
